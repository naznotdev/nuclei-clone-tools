# ============================================================
# MATCHER HELPERS
# ============================================================
def W(*words):
    """Word matcher (body)."""
    return {"type": "word", "part": "body", "words": list(words)}

def WH(*words):
    """Word matcher (header)."""
    return {"type": "word", "part": "header", "words": list(words)}

def R(pattern):
    """Regex matcher (body)."""
    return {"type": "regex", "part": "body", "regex": [pattern]}

def S(*statuses):
    """Status matcher."""
    return {"type": "status", "status": list(statuses)}

def _make_id(path, idx):
    slug = re.sub(r'[^a-z0-9]+', '-', path.strip('/').lower()).strip('-')
    return f"nz-{idx+1:03d}-{slug[:40]}"


def generate_templates_from_checks():
    templates = []
    for idx, entry in enumerate(SENSITIVE_CHECKS):
        if len(entry) == 5:
            path, severity, name, desc, matchers = entry
            condition = "or"
        else:
            path, severity, name, desc, matchers, condition = entry
        tid = _make_id(path, idx)
        tags = "exposure,misconfig,nuclei-clone"
        if severity in ("critical", "high"):
            tags += ",intrusive"
        templates.append({
            "id": tid,
            "info": {
                "name": name,
                "author": "nazreal",
                "severity": severity,
                "description": desc,
                "tags": tags,
            },
            "http": [{
                "method": "GET",
                "path": ["{{BaseURL}}" + path],
                "redirects": False,
                "max-redirects": 2,
                "matchers-condition": condition,
                "matchers": matchers,
            }],
        })
    return templates

class NucleiLite:
    def __init__(self, target=None, targets=None, templates_dir=None,
                 severity_filter=None, tag_filter=None, concurrency=20,
                 rate_limit=50, timeout=10, retries=1, headers=None,
                 proxy=None, verbose=False, silent=False, no_color=False,
                 output_jsonl=None, output_md=None, output_html=None,
                 show_stats=False):
        self.targets = []
        if targets:
            self.targets.extend(targets)
        if target:
            self.targets.append(target)
        self.targets = [self._normalize(t) for t in self.targets if t]

        self.templates = []
        self.templates_dir = templates_dir
        self.severity_filter = set(s.lower() for s in severity_filter) if severity_filter else None
        self.tag_filter = set(t.lower() for t in tag_filter) if tag_filter else None
        self.concurrency = max(1, concurrency)
        self.rate_limit = max(1, rate_limit)
        self.timeout = timeout
        self.retries = max(0, retries)
        self.custom_headers = headers or {}
        self.proxy = proxy
        self.verbose = verbose
        self.silent = silent
        self.no_color = no_color
        self.output_jsonl = output_jsonl
        self.output_md = output_md
        self.output_html = output_html
        self.show_stats = show_stats

        self.session = requests.Session()
        self.session.headers.update({
            "User-Agent": "Mozilla/5.0 (Linux; Android 13; Termux) NucleiClone/1.10",
            "Accept": "*/*",
            "Accept-Language": "en-US,en;q=0.9",
        })
        self.session.headers.update(self.custom_headers)
        if proxy:
            self.session.proxies.update({"http": proxy, "https": proxy})

        self.lock = threading.Lock()
        self.results = []
        self._last_request_time = 0.0
        self._rate_lock = threading.Lock()
        self._request_count = 0
        self._start_time = None

    # ---------- helpers ----------
    @staticmethod
    def _normalize(url):
        url = url.strip()
        if not url:
            return ""
        if not url.startswith(("http://", "https://")):
            url = "https://" + url
        return url.rstrip("/")

    def _c(self, color, text):
        if self.no_color:
            return text
        return f"{color}{text}{Style.RESET_ALL}"

    def _log(self, msg, color=None):
        if self.silent:
            return
        with self.lock:
            if color:
                print(self._c(color, msg))
            else:
                print(msg)

    # ---------- template loading ----------
    def load_templates(self):
        loaded = []
        if self.templates_dir and os.path.isdir(self.templates_dir):
            for root, _, files in os.walk(self.templates_dir):
                for f in files:
                    if f.endswith((".yaml", ".yml")):
                        p = os.path.join(root, f)
                        try:
                            with open(p, "r", encoding="utf-8", errors="ignore") as fh:
                                for doc in yaml.safe_load_all(fh):
                                    if self._valid_template(doc):
                                        loaded.append(doc)
                        except Exception:
                            continue
        if not loaded:
            loaded = generate_templates_from_checks()

        if self.severity_filter:
            loaded = [t for t in loaded
                      if str(t.get("info", {}).get("severity", "")).lower() in self.severity_filter]

        if self.tag_filter:
            filtered = []
            for t in loaded:
                tags = t.get("info", {}).get("tags", "")
                if isinstance(tags, str):
                    tags = [x.strip().lower() for x in tags.split(",") if x.strip()]
                else:
                    tags = [str(x).lower() for x in tags]
                if any(tg in tags for tg in self.tag_filter):
                    filtered.append(t)
            loaded = filtered

        loaded.sort(key=lambda t: -SEV_ORDER.get(
            str(t.get("info", {}).get("severity", "unknown")).lower(), 0))
        self.templates = loaded
        return loaded

    @staticmethod
    def _valid_template(t):
        return (isinstance(t, dict) and "id" in t and "info" in t
                and ("http" in t or "requests" in t))

    # ---------- rate limit ----------
    def _wait_rate(self):
        with self._rate_lock:
            now = time.time()
            interval = 1.0 / self.rate_limit
            wait = interval - (now - self._last_request_time)
            if wait > 0:
                time.sleep(wait)
            self._last_request_time = time.time()
            self._request_count += 1

    # ---------- matchers ----------
    def _get_part(self, response, part):
        if part == "header":
            return "\n".join(f"{k}: {v}" for k, v in response.headers.items())
        if part == "all":
            return response.text + "\n" + "\n".join(f"{k}: {v}" for k, v in response.headers.items())
        return response.text or ""

    def _eval_dsl(self, expr, response):
        try:
            for bad in ["__", "import", "exec", "eval", "open(", "os.", "sys.", "subprocess"]:
                if bad in expr:
                    return False
            env = {
                "status_code": response.status_code,
                "body": response.text or "",
                "content_length": len(response.content),
                "header": dict(response.headers),
                "true": True, "false": False, "null": None,
                "contains": lambda s, sub: sub in (s or ""),
                "icontains": lambda s, sub: sub.lower() in (s or "").lower(),
                "regex": lambda pat, s: bool(re.search(pat, s or "")),
                "md5": lambda s: hashlib.md5((s or "").encode()).hexdigest(),
                "sha256": lambda s: hashlib.sha256((s or "").encode()).hexdigest(),
            }
            return bool(eval(expr, {"__builtins__": {}}, env))
        except Exception:
            return False

    def _match(self, matchers, block_cond, response):
        if not matchers:
            return True
        results = []
        for m in matchers:
            mtype = str(m.get("type", "")).lower()
            part = str(m.get("part", "body")).lower()
            negative = bool(m.get("negative", False))
            matcher_cond = str(m.get("condition", "or")).lower()

            if mtype == "status":
                st = m.get("status", [])
                if isinstance(st, int):
                    st = [st]
                ok = response.status_code in st

            elif mtype == "word":
                words = m.get("words", [])
                if isinstance(words, str):
                    words = [words]
                ci = m.get("case-insensitive", True)
                text = self._get_part(response, part)
                if ci:
                    text = text.lower()
                    words = [str(w).lower() for w in words]
                ok = all(w in text for w in words) if matcher_cond == "and" else any(w in text for w in words)

            elif mtype == "regex":
                pats = m.get("regex", [])
                if isinstance(pats, str):
                    pats = [pats]
                text = self._get_part(response, part)
                if matcher_cond == "and":
                    ok = all(bool(re.search(p, text)) for p in pats)
                else:
                    ok = any(bool(re.search(p, text)) for p in pats)

            elif mtype == "size":
                sizes = m.get("size", [])
                if isinstance(sizes, int):
                    sizes = [sizes]
                ok = len(response.content) in sizes

            elif mtype == "dsl":
                exprs = m.get("dsl", [])
                if isinstance(exprs, str):
                    exprs = [exprs]
                ok = any(self._eval_dsl(e, response) for e in exprs)

            elif mtype == "binary":
                ok = False  # not implemented (needs raw socket)
            else:
                ok = False

            if negative:
                ok = not ok
            results.append(ok)

        return all(results) if block_cond == "and" else any(results)

    # ---------- extractors ----------
    def _extract(self, extractors, response):
        out = []
        for ext in extractors or []:
            etype = str(ext.get("type", "")).lower()
            part = str(ext.get("part", "body")).lower()
            text = self._get_part(response, part)
            if etype == "regex":
                pats = ext.get("regex", [])
                if isinstance(pats, str):
                    pats = [pats]
                for p in pats:
                    try:
                        for m in re.finditer(p, text):
                            out.append({"type": "regex", "pattern": p, "value": m.group(0)[:120]})
                    except re.error:
                        continue
            elif etype == "kval":
                keys = ext.get("kval", [])
                if isinstance(keys, str):
                    keys = [keys]
                for k, v in response.headers.items():
                    if k.lower() in [x.lower() for x in keys]:
                        out.append({"type": "kval", "name": k, "value": v})
        return out

    # ---------- substitution ----------
    def _substitute(self, text, target):
        if not isinstance(text, str):
            return str(text)
        host = urlparse(target).netloc
        rand = "".join(random.choices(string.ascii_lowercase + string.digits, k=10))
        return (text.replace("{{BaseURL}}", target)
                    .replace("{{Hostname}}", urlparse(target).hostname or "")
                    .replace("{{Host}}", host)
                    .replace("{{RootURL}}", target)
                    .replace("{{RandomString}}", rand))

    # ---------- execute one template ----------
    def _execute(self, template, target):
        tid = template.get("id", "unknown")
        info = template.get("info", {})
        http_blocks = template.get("http") or template.get("requests") or []

        for block in http_blocks:
            method = str(block.get("method", "GET")).upper()
            paths = block.get("path", [])
            if isinstance(paths, str):
                paths = [paths]
            block_headers = block.get("headers", {})
            body = block.get("body")
            matchers = block.get("matchers", [])
            cond = str(block.get("matchers-condition", "or")).lower()
            extractors = block.get("extractors", [])
            follow = bool(block.get("redirects", False))

            for path_tpl in paths:
                path = self._substitute(path_tpl, target)
                url = urljoin(target + "/", path.lstrip("/"))

                resp = None
                for attempt in range(self.retries + 1):
                    self._wait_rate()
                    try:
                        resp = self.session.request(
                            method=method, url=url, headers=block_headers,
                            data=body, timeout=self.timeout,
                            verify=False, allow_redirects=follow,
                        )
                        break
                    except Exception as e:
                        if attempt >= self.retries:
                            if self.verbose:
                                self._log(f"[ERR] {tid} -> {url} : {e}", Fore.RED)
                            resp = None

                if resp is None:
                    continue

                if self._match(matchers, cond, resp):
                    extracted = self._extract(extractors, resp)
                    self._report(template, target, url, resp, extracted)
                    break

    # ---------- reporting ----------
    def _report(self, template, target, url, response, extracted=None):
        info = template.get("info", {})
        severity = str(info.get("severity", "unknown")).lower()
        tid = template.get("id", "unknown")
        name = info.get("name", tid)
        color = SEV_COLOR.get(severity, Fore.WHITE)
        badge = SEV_BADGE.get(severity, "[ UNKNOWN  ]")
        ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

        line = (f"{self._c(color, badge)} "
                f"[{self._c(Fore.CYAN, tid)}] "
                f"[{url}] "
                f"[{self._c(Fore.GREEN, name)}]")

        with self.lock:
            self.results.append({
                "timestamp": ts,
                "target": target,
                "severity": severity,
                "template_id": tid,
                "name": name,
                "url": url,
                "status_code": response.status_code,
                "content_length": len(response.content),
                "extracted": extracted or [],
            })
            if not self.silent:
                print(line)
            if self.output_jsonl:
                try:
                    with open(self.output_jsonl, "a", encoding="utf-8") as fh:
                        fh.write(json.dumps(self.results[-1]) + "\n")
                except Exception:
                    pass

    # ---------- run ----------
    def _print_header(self):
        if self.silent:
            return
        print(self._c(Fore.CYAN, BANNER))
        print(self._c(Fore.YELLOW, f"  Version  : 1.10 (Deep & Complete)"))
        print(self._c(Fore.YELLOW, f"  Targets  : {len(self.targets)}"))
        print(self._c(Fore.YELLOW, f"  Templates: {len(self.templates)}"))
        print(self._c(Fore.YELLOW, f"  Threads  : {self.concurrency}  |  Rate: {self.rate_limit}/s  |  Timeout: {self.timeout}s"))
        if self.severity_filter:
            print(self._c(Fore.YELLOW, f"  Severity : {', '.join(sorted(self.severity_filter))}"))
        print(self._c(Fore.WHITE, "-" * 78))

    def _print_stats(self, elapsed):
        if self.silent:
            return
        counts = {}
        for r in self.results:
            counts[r["severity"]] = counts.get(r["severity"], 0) + 1
        print(self._c(Fore.WHITE, "-" * 78))
        print(self._c(Fore.CYAN, "=== SCAN STATISTICS ==="))
        print(f"  Targets        : {len(self.targets)}")
        print(f"  Templates      : {len(self.templates)}")
        print(f"  Requests sent  : {self._request_count}")
        print(f"  Total findings : {len(self.results)}")
        for sev in ["critical", "high", "medium", "low", "info", "unknown"]:
            if sev in counts:
                print(f"    {self._c(SEV_COLOR[sev], SEV_BADGE[sev])} : {counts[sev]}")
        print(f"  Duration       : {elapsed:.2f}s")

    def _write_markdown(self, elapsed):
        if not self.output_md:
            return

    def run(self):
        if not self.templates:
            self._log("[!] Tidak ada template yang dimuat.", Fore.RED)
            return
        if not self.targets:
            self._log("[!] Tidak ada target.", Fore.RED)
            return

        self._start_time = time.time()
        self._print_header()

        def worker(tpl, tgt):
            try:
                self._execute(tpl, tgt)
            except Exception as e:
                if self.verbose:
                    self._log(f"[!] {tpl.get('id')} on {tgt} crashed: {e}", Fore.RED)

        with concurrent.futures.ThreadPoolExecutor(max_workers=self.concurrency) as ex:
            futures = [ex.submit(worker, t, tgt)
                       for tgt in self.targets for t in self.templates]
            for f in concurrent.futures.as_completed(futures):
                try:
                    f.result()
                except Exception:
                    pass

        elapsed = time.time() - self._start_time

        if self.show_stats:
            self._print_stats(elapsed)
        else:
            self._log(f"\n[+] Selesai dalam {elapsed:.2f}s. {len(self.results)} temuan.", Fore.GREEN)

        self._write_markdown(elapsed)
        self._write_html(elapsed)

        if self.output_jsonl:
            self._log(f"[+] JSONL disimpan ke: {self.output_jsonl}", Fore.CYAN)
        if self.output_md:
            self._log(f"[+] Markdown disimpan ke: {self.output_md}", Fore.CYAN)
        if self.output_html:
            self._log(f"[+] HTML disimpan ke: {self.output_html}", Fore.CYAN)

    p.add_argument("-v", "--verbose", action="store_true", help="Verbose error")
    p.add_argument("--silent", action="store_true", help="Mode senyap (hanya laporan file)")
    p.add_argument("--no-color", action="store_true", help="Nonaktifkan warna")
    p.add_argument("--stats", action="store_true", help="Tampilkan statistik di akhir")
    p.add_argument("-tl", "--list-templates", action="store_true", help="Tampilkan semua template yang tersedia")
    p.add_argument("--version", action="store_true", help="Tampilkan versi")
    return p.parse_args()


def parse_headers(raw_list):
    headers = {}
    for item in raw_list:
        if ":" in item:
            k, v = item.split(":", 1)
            headers[k.strip()] = v.strip()
    return headers


def load_target_file(path):
    targets = []
    try:
        with open(path, "r", encoding="utf-8", errors="ignore") as fh:
            for line in fh:
                line = line.strip()
                if line and not line.startswith("#"):
                    targets.append(line)
    except Exception as e:
        print(f"[!] Gagal baca target file: {e}", file=sys.stderr)
    return targets


def list_templates_and_exit():
    templates = generate_templates_from_checks()
    print(f"{Fore.CYAN}[+] Total template internal: {len(templates)}{Style.RESET_ALL}")
    print(f"{'SEVERITY':<10} {'ID':<45} NAME")
    print("-" * 100)
    for t in templates:
        sev = t["info"]["severity"].lower()
        color = SEV_COLOR.get(sev, Fore.WHITE)
        print(f"{color}{sev.upper():<10}{Style.RESET_ALL} {t['id']:<45} {t['info']['name']}")
    sys.exit(0)


def main():
    args = parse_args()

    if args.version:
        print("Nuclei Clone v1.10 — Deep & Complete Edition")
        sys.exit(0)

    if args.list_templates:
        if args.no_color:
            Fore.CYAN = Fore.WHITE = Style.RESET_ALL = ""
        list_templates_and_exit()

    targets = []
    if args.target_list:
        targets.extend(load_target_file(args.target_list))
    if args.url:
        targets.append(args.url)

    if not targets:
        print(f"{Fore.RED}[!] Butuh -u <URL> atau -l <file>{Style.RESET_ALL}")
        sys.exit(1)

    sev_filter = None
    if args.severity:
        sev_filter = [s.strip().lower() for s in args.severity.split(",") if s.strip()]

    tag_filter = None
    if args.tags:
        tag_filter = [t.strip().lower() for t in args.tags.split(",") if t.strip()]

    headers = parse_headers(args.header)

    scanner = NucleiLite(
        targets=targets,
        templates_dir=args.templates,
        severity_filter=sev_filter,
        tag_filter=tag_filter,
        concurrency=args.concurrency,
        rate_limit=args.rate_limit,
        timeout=args.timeout,
        retries=args.retry,
        headers=headers,
        proxy=args.proxy,
        verbose=args.verbose,
        silent=args.silent,
        no_color=args.no_color,
        output_jsonl=args.output,
        output_md=args.markdown,
        output_html=args.html,
        show_stats=args.stats,
    )
    scanner.load_templates()
    scanner.run()


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print(f"\n{Fore.YELLOW}[!] Dihentikan oleh pengguna.{Style.RESET_ALL}")
        sys.exit(0)
